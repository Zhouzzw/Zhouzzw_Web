/* LQR 平衡控制器 · 每 1ms 中断调用 */
void lqr_balance_loop(void)
{
    update_sensors();              // IMU + 编码器
    x_k[0] = gimbal_pitch;              // 俯仰角
    x_k[1] = pitch_rate;               // 俯仰角速度
    x_k[2] = wheel_vel;                // 轮速

    u_k = K_lqr · x_k;                 // LQR 最优控制律
    set_motor_duty(&motor[0], u_k);
    set_motor_duty(&motor[1], -u_k);
}

RoboMaster 全向轮步兵机器人
